%Define la longitud y la escala para la línea base
\newcommand{\headingBaseline}{12}
\newcommand{\headingBaselineDiv}{10}
\newlength{\chapterFontSize}
\newlength{\sectionFontSize}
\newlength{\subsectionFontSize}
\newlength{\chapterBaseline}
\newlength{\sectionBaseline}
\newlength{\subsectionBaseline}

%Tamaño de la letra del documento
\setlength{\chapterFontSize}{14pt}
\setlength{\sectionFontSize}{12pt}
\setlength{\subsectionFontSize}{12pt}

%% automatic computation for baseline, rule of thumb is 1.2
\setlength{\chapterBaseline}{ \chapterFontSize * \headingBaseline / \headingBaselineDiv}
\setlength{\sectionBaseline}{ \sectionFontSize * \headingBaseline / \headingBaselineDiv}
\setlength{\subsectionBaseline}{ \subsectionFontSize * \headingBaseline / \headingBaselineDiv}

%Título 1 – Arial, 14, Bold, Espaciado 1.5 entre párrafos, Justificado
\titleformat{\chapter}[display]
    {\normalfont\bfseries\fontsize{14pt}{18pt}\selectfont}{Capítulo \thechapter}{20pt}{\bfseries}
\titlespacing*{\chapter}{0pt}{1.0em}{1.0em}

% Título 2 – Arial, 12, Bold, Espaciado 1.5 entre párrafos, Justificado
\titleformat{\section}[hang]
    {\normalfont\bfseries\fontsize{12pt}{15pt}\selectfont}{\thesection}{12pt}{}
\titlespacing*{\section}{0pt}{1.0em}{1.0em}

%Título 3 – Arial, 11, Bold, Espaciado 1.5 entre párrafos, Justificado
\titleformat{\subsection}[hang]
    {\normalfont\bfseries\fontsize{11pt}{13pt}\selectfont}{\thesubsection}{10pt}{}
%\titlespacing*{\subsection}{0pt}{0pt}{1.5em}
\titlespacing*{\subsection}{0pt}{1.0em}{1.0em}

%Título 4 – Arial, 11, Itálica, Espaciado 1.5 entre párrafos, Justificado
\titleformat{\subsubsection}[hang]
    {\normalfont\itshape\fontsize{11pt}{13pt}\selectfont}{\thesubsubsection}{8pt}{}
%\titlespacing*{\subsubsection}{0pt}{0pt}{1.5em}
\titlespacing*{\subsubsection}{0pt}{0pt}{0em}
